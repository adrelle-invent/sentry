## Ingestion API Payload Schema (FastAPI Gateway)
This defines the contract between external client applications and your FastAPI ingestion endpoint (POST /api/v1/ingest).

```json
{
  "$schema": "http://json-schema.org/draft-07/schema#",
  "title": "LogIngestPayload",
  "type": "object",
  "required": ["timestamp", "level", "service", "message"],
  "properties": {
    "timestamp": {
      "type": "string",
      "format": "date-time",
      "description": "ISO-8601 UTC timestamp when the event occurred on the client."
    },
    "level": {
      "type": "string",
      "enum": ["DEBUG", "INFO", "WARNING", "ERROR", "CRITICAL"],
      "description": "Standardized log severity level."
    },
    "service": {
      "type": "string",
      "maxLength": 128,
      "description": "The name of the microservice or module emitting the log."
    },
    "message": {
      "type": "string",
      "maxLength": 8192,
      "description": "The primary log message or exception description."
    },
    "stack_trace": {
      "type": ["string", "null"],
      "maxLength": 32768,
      "description": "Raw unformatted stack trace string if an exception occurred."
    },
    "environment": {
      "type": "string",
      "default": "production",
      "description": "Deployment environment (e.g., production, staging, development)."
    },
    "tags": {
      "type": "object",
      "additionalProperties": {
        "type": "string"
      },
      "description": "Arbitrary key-value string tags for filtering (e.g., region, version)."
    }
  }
}
```
_Note: The tenant identifier is extracted securely from the incoming HTTP header (X-Tenant-ID) rather than trusted inside the JSON body._

# Redis Stream Event Message Schema
This defines the internal JSON payload pushed into Redis Streams (Step 4) after FastAPI validates the request and passes rate-limiting checks.

```json
{
  "$schema": "http://json-schema.org/draft-07/schema#",
  "title": "StreamLogEvent",
  "type": "object",
  "required": [
    "event_id",
    "tenant_id",
    "received_at",
    "signature_hash",
    "payload"
  ],
  "properties": {
    "event_id": {
      "type": "string",
      "format": "uuid",
      "description": "Unique UUIDv4 generated by the gateway for tracing."
    },
    "tenant_id": {
      "type": "string",
      "description": "Multi-tenant isolation key extracted from request headers."
    },
    "received_at": {
      "type": "string",
      "format": "date-time",
      "description": "UTC timestamp when the ingestion gateway accepted the request."
    },
    "signature_hash": {
      "type": "string",
      "maxLength": 64,
      "description": "SHA-256 deterministic signature hash computed from normalized stack traces/error messages for grouping."
    },
    "payload": {
      "$ref": "#/definitions/LogIngestPayload"
    }
  }
}
```

# Database Table Structures (PostgreSQL + pgvector)
This DDL script sets up your partitioned hot storage tables, raw log indices, and the isolated vector summary table governed by your daily cron pipeline.

```sql
-- Enable the pgvector extension (must be run as superuser or database owner)
CREATE EXTENSION IF NOT EXISTS vector;

-----------------------------------------------------------------
-- 1. HOT LOGS TABLE (Partitioned by Tenant & Time for Dashboard)
-----------------------------------------------------------------
CREATE TABLE hot_log_aggregates (
    id BIGSERIAL,
    tenant_id VARCHAR(64) NOT NULL,
    signature_hash VARCHAR(64) NOT NULL,
    service VARCHAR(128) NOT NULL,
    log_level VARCHAR(16) NOT NULL,
    event_count INT NOT NULL DEFAULT 1,
    first_seen TIMESTAMP WITH TIME ZONE NOT NULL,
    last_seen TIMESTAMP WITH TIME ZONE NOT NULL,
    sample_message TEXT NOT NULL,
    sample_stack_trace TEXT,
    created_at TIMESTAMP WITH TIME ZONE DEFAULT NOW(),
    PRIMARY KEY (tenant_id, id)
) PARTITION BY RANGE (tenant_id);

-- Create index for quick dashboard queries and signature lookups
CREATE INDEX idx_hot_logs_signature 
ON hot_log_aggregates (tenant_id, signature_hash, last_seen DESC);


-----------------------------------------------------------------
-- 2. VECTOR SUMMARY TABLE (Step 14 Target Layer)
-----------------------------------------------------------------
CREATE TABLE tenant_vector_summaries (
    id BIGSERIAL PRIMARY KEY,
    tenant_id VARCHAR(64) NOT NULL,
    granularity VARCHAR(16) NOT NULL CHECK (granularity IN ('daily', 'monthly')),
    summary_text TEXT NOT NULL,                -- Narrative text generated by local Gemma SLM
    start_date DATE NOT NULL,                  -- Batch window start
    end_date DATE NOT NULL,                    -- Batch window end
    total_error_count INT NOT NULL,            -- Deterministic metric backing the summary
    embedding vector(768) NOT NULL,            -- Output from embeddinggemma-300m
    created_at TIMESTAMP WITH TIME ZONE DEFAULT NOW()
);

-- Index for multi-tenant date range filtering
CREATE INDEX idx_vector_metadata 
ON tenant_vector_summaries (tenant_id, granularity, start_date DESC);

-- HNSW Vector Index for blazing fast cosine similarity search (RAG Layer)
CREATE INDEX idx_vector_cosine_hnsw 
ON tenant_vector_summaries 
USING hnsw (embedding vector_cosine_ops)
WITH (m = 16, ef_construction = 64);
```
